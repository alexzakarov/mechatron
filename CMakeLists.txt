cmake_minimum_required(VERSION 3.21)
project(mechatron VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_compile_definitions(NOMINMAX)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Output directories
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# OpenGL
find_package(OpenGL REQUIRED)

# Find vcpkg packages
find_package(glfw3 CONFIG REQUIRED)
find_package(GLEW CONFIG REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(glm CONFIG REQUIRED)

# Jolt Physics (optional)
set(JOLT_PHYSICS_ROOT "" CACHE PATH "Path to JoltPhysics root directory")
if(JOLT_PHYSICS_ROOT)
    add_subdirectory(${JOLT_PHYSICS_ROOT} ${CMAKE_BINARY_DIR}/jolt_physics)
endif()

# Plugin options (all enabled by default)
option(MECHATRON_PLUGIN_MECH_MACHINE_ELEMENTS "Enable Machine Elements Plugin" ON)
option(MECHATRON_PLUGIN_ELEC_PASSIVE "Enable Passive Electronics Plugin" ON)
option(MECHATRON_PLUGIN_ELEC_SEMICONDUCTOR "Enable Semiconductor Electronics Plugin" ON)
option(MECHATRON_PLUGIN_ELEC_POWER "Enable Power Electronics Plugin" ON)
option(MECHATRON_PLUGIN_SOFT_MCU_AVR "Enable AVR MCU Plugin" ON)
option(MECHATRON_PLUGIN_SOFT_CONTROL "Enable Control Algorithms Plugin" ON)
option(MECHATRON_PLUGIN_MULTI_THERMAL "Enable Thermal Physics Plugin" ON)
option(MECHATRON_PLUGIN_MULTI_MAGNETIC "Enable Magnetic Physics Plugin" ON)

# Subdirectories - Core (must be first)
add_subdirectory(src/core)

# Plugin directories (modular architecture, conditional)
if(MECHATRON_PLUGIN_MECH_MACHINE_ELEMENTS)
    add_subdirectory(src/plugins/mechanics/machine_elements)
    target_compile_definitions(mechatron_core PUBLIC MECH_MACHINE_ELEMENTS_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_ELEC_PASSIVE)
    add_subdirectory(src/plugins/electronics/passive)
    target_compile_definitions(mechatron_core PUBLIC ELEC_PASSIVE_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_ELEC_SEMICONDUCTOR)
    add_subdirectory(src/plugins/electronics/semiconductor)
    target_compile_definitions(mechatron_core PUBLIC ELEC_SEMICONDUCTOR_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_ELEC_POWER)
    add_subdirectory(src/plugins/electronics/power)
    target_compile_definitions(mechatron_core PUBLIC ELEC_POWER_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_SOFT_MCU_AVR)
    add_subdirectory(src/plugins/software/mcu_avr)
    target_compile_definitions(mechatron_core PUBLIC SOFT_MCU_AVR_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_SOFT_CONTROL)
    add_subdirectory(src/plugins/software/control)
    target_compile_definitions(mechatron_core PUBLIC SOFT_CONTROL_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_MULTI_THERMAL)
    add_subdirectory(src/plugins/multiphysics/thermal)
    target_compile_definitions(mechatron_core PUBLIC MULTI_THERMAL_ENABLED=1)
endif()
if(MECHATRON_PLUGIN_MULTI_MAGNETIC)
    add_subdirectory(src/plugins/multiphysics/magnetic)
    target_compile_definitions(mechatron_core PUBLIC MULTI_MAGNETIC_ENABLED=1)
endif()

# Instruments plugin (always enabled)
add_subdirectory(src/plugins/instruments)
target_compile_definitions(mechatron_core PUBLIC INSTRUMENT_ENABLED=1)

# Support modules (non-plugin, legacy integration)
add_subdirectory(src/physics)
add_subdirectory(src/mcu)
add_subdirectory(src/cad)
add_subdirectory(src/electronics)
add_subdirectory(src/io)
add_subdirectory(src/sensors)
add_subdirectory(src/actuators)
add_subdirectory(src/control)
add_subdirectory(src/renderer)
add_subdirectory(src/ui)

# Main application
add_subdirectory(src/app)

# Testing
option(MECHATRON_BUILD_TESTS "Build tests" OFF)
if(MECHATRON_BUILD_TESTS)
    enable_testing()
    find_package(GTest CONFIG REQUIRED)
    add_subdirectory(tests)
endif()

# Examples
option(MECHATRON_BUILD_EXAMPLES "Build example programs" ON)
if(MECHATRON_BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
